# Inheritance (상속)
 - 기존 클래스의 기능을 물려받아 새로운 클래스를 만드는 기능
 - 부모 클래스, 자식 클래스
	> 자식 클래스는 부모의 기능 + 자신만의 기능 사용 가능
~~~	
class 부모클래스
{
	멤버 변수
	생성자
	멤버 함수
}
~~~
~~~
Ex. class Animal
{
	public string Name; // 멤버 변수
	public Animal(string name) { Name = name; } // 생성자
}
~~~

// 자식 클래스는 부모 클래스의 상속을 받는다 	--	
class 자식클래스 : 부모클래스				--	// 멤버 함수
{								--	public void Eat() {CW($"{Name}이(가) 밥을 먹습니다.")}
								--  }
}								--
								--  class Dog : Animal
// 자식 클래스 변수로는 부모 클래스 객체를	--  {
// 가리킬 수 없다.					--	// 자식 클래스는 반드시 부모 클래스의 생성자 중 하나를 사용
// 자식클래스 A = new 부모클래스;	 // X		--	public Dog(string name) : base(name) {}
								--	// 부모의 멤버 변수 사용 가능
// 자식은 부모클래스의 기능을 사용 할 수 있다.--	public void Bark() {CW($"{Name}이(가) 짖습니다.")}

상속을 쓰는 이유
 1) 새 클래스를 만들 때 같은 코드를 여러 번 작성하지 않아도 된다.
 2) 공통 기능을 한 곳에서(부모 클래스에서) 관리할 수 있다.
 3) 유사한 기능을 갖는 클래스들을 쉽게 만들 수 있다.
