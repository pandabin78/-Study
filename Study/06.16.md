```cs
using System.Collections;
using UnityEngine;

public class UnitTest_Coroutine : MonoBehaviour
{
    [SerializeField]
    private int frame = 180;

    private void Start()
    {
        
    }

    private void Update()
    {
        if (Input.GetKeyDown(KeyCode.Alpha1))
        {
            for (int i = 0; i < frame; i++)
            {
                print($"{i} Frame");
            }
        }

        if (Input.GetKeyDown(KeyCode.Alpha2))
        {
            StartCoroutine(Co_Update());
        }
    }

    private IEnumerator Co_Update()
    {
        print("현재 프레임");

        for(int i = 0; i < frame; i++)
        {
            print($"{i} Frame");
            yield return null;
        }
            

        print("다음 프레임");
    }
}
```
