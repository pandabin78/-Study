# 06.04 수업 내용 정리

토글 버튼 생성
{ // &= and기호
        bool bCheck = true;
        bCheck &= Input.GetKey(KeyCode.LeftControl);
        bCheck &= Input.GetKey(KeyCode.LeftShift);

        if (bCheck)
        {
            if (Input.GetKeyDown(KeyCode.Alpha1))
            {
                bCheck &= bCheetMode = !bCheetMode;
            }
            else if (Input.GetKeyDown(KeyCode.Alpha2))
            {
                blockMoveSpeed = 0.0f;
            }
            else if (Input.GetKeyDown(KeyCode.Minus))
            {
                blockMoveSpeed -= 1.0f;
                blockMoveSpeed = Mathf.Clamp(blockMoveSpeed, 0.0f, 20.0f);
            }
            else if (Input.GetKeyDown(KeyCode.Equals))
            {
                blockMoveSpeed += 1.0f;
                blockMoveSpeed = Mathf.Clamp(blockMoveSpeed, 0.0f, 20.0f);
            }
        }//if (bCheck)

    }

    삭제
    Destroy(gameObject, 2.0f);
