# 06.02 수업 내용 정리

```cs
public class UnitTest_Collision : MonoBehaviour
{
    private void OnCollisionEnter2D(Collision2D collision)
    {
        ContactPoint2D contact = collision.GetContact(0);
        print($"collision : {contact.normal}");

        Debug.DrawRay(transform.position, transform.up, Color.green, 10.0f);
        Debug.DrawRay(transform.position, transform.right, Color.red, 10.0f);
        Debug.DrawRay(transform.position, contact.normal, Color.blue, 10.0f);

        float dotUp = Vector2.Dot(transform.up, -contact.normal);
        print($"dotUp : {dotUp}");

        float dotRight = Vector2.Dot(transform.right, -contact.normal);
        print($"dotRight : {dotRight}");

        if(Mathf.Abs(dotRight) > Mathf.Abs(dotUp))
        {
            string side = dotRight > 0 ? "right" : "left";
            print(side);
        }
        else
        {
            string side = dotUp > 0 ? "top" : "bottom";// ?(삼항) 연산자 앞에가 맞으면 top, 틀리면, bottom
            print(side);
        }

    }
}
```

#### 사용된 코드 정리
1. Debug.DrawRay(transform.position, transform.up, Color.green, 10.0f);
    > 계발자 시점에만 보이는 가상에 선을 그림\
    (시작 위치, 방향, 색상, 유지시간)

2. float dotUp = Vector2.Dot(transform.up, -contact.normal);
    > 본인에 업 방향과 충돌대상에 뒤집은 충돌방향 내적을 구해서 dotup에 넣음\
    (본인 방향, -충돌 방향)

3. if(Mathf.Abs(dotRight) > Mathf.Abs(dotUp))
    > 만약 옆면 내적이 상하면 내적보다 크면

4. string side = dotRight > 0 ? "right" : "left";
    > 옆면 값이 0보다 큰게 맞다면 "right", 아니라면 "left"

5. else string side = dotUp > 0 ? "top" : "bottom";
    > Up 충돌값이 Right 값보다 크다면 if 문이 틀려 else로 빠지고\
    추가로 top, bottom으로 구분.
