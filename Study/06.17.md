# 06.17 수업 내용 정리

realtimeSinceStartup 일시정지를 해도 움직이는 코드
예 : 로딩화면 UI

```cs
private IEnumerator Co_Time3(float time)
    {
        image.fillAmount = 0.0f;
        //ui 크기 를 0으로

        float startTime = Time.time;
        //startTime에 게임 시작 값 넣음

        while (true)
        {
            yield return new WaitForSeconds(0.1f);
            // yield return은 Coroutine에 사용되는 리턴값
            // new WaitForSeconds(0.1f);은 0.1f뒤에 리턴되라

            float current = Time.time - startTime;
            //current(흐른 시간)에 현재 시간 - 시작시간

            image.fillAmount = current / time;
            // UI이미지를 흐른시간 / 적용한 시간(0.1f) 으로 값을 가둔것.

            if (current > time)
                break;
        }
    }

    private IEnumerator Co_Time4(float time)
    {
        image.fillAmount = 0.0f;

        float startTime = Time.realtimeSinceStartup;

        while (true)
        {
            yield return new WaitForSecondsRealtime(0.1f);

            float current = Time.realtimeSinceStartup - startTime;

            image.fillAmount = current / time;

            if (current > time)
                break;
        }
    }
```

예 : player = transform.root.GetComponent<Player>();

유저의 State확인을 위한 UI설정
```cs
using TMPro;
using UnityEngine;

public class Player_TextState : MonoBehaviour
{
    private TextMeshProUGUI tmp;

    private Player player;
    private Animator animator;
    
    private void Awake()
    {
        tmp = GetComponent<TextMeshProUGUI>();

        player = transform.root.GetComponent<Player>();
        animator = transform.root.GetComponent<Animator>();
    }

    private void Start()
    {
        
    }

    private void Update()
    {
        bool jump = animator.GetBool("bJump");

        string textJump = $"Jump : <color=#FF0000>{jump}</color>\n";
        string textCoyote = $"Coyote : <color=#FF0000>0.00</color>\n";
        string textBuffer = $"Buffer : <color=#FF0000>0.00</color>";

        tmp.text = textJump + textCoyote + textBuffer;
    }
}
```
