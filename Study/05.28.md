private void Update_Raycast()
    {
        Vector3 start = Vector3.zero;
        start += transform.position;
        start.y += transform.localScale.y * 0.25f;

        Vector2 point = Vector2.zero;
        Vector2 normal = Vector2.zero;
        Collider2D collider = null;

        if (Raycasting(start, Vector3.right, ref collider, ref point, ref normal))
        {
            print(normal);
            print(-collider.transform.right);
            print(Vector2.Dot(normal, -collider.transform.right));
        }
        Raycasting(start, Vector3.down, ref collider, ref point, ref normal);
    }

    private bool Raycasting(Vector3 start, Vector3 direction, ref Collider2D collider, ref Vector2 hitPoint, ref Vector2 hitNormal)
    {
        int layer = ~LayerMask.GetMask("Player");
        //print(layer);        

        RaycastHit2D hit = Physics2D.Raycast(start, direction, 1, layer);// Physics2D rigidbody가 붙어있어야 한다.

        bool bCheck = true;
        bCheck &= hit.collider is not null;
        bCheck &= hit.collider is BoxCollider2D;

        if (bCheck)
        {
            hitPoint = hit.point;
            hitNormal = hit.normal;
            collider = hit.collider;

            Debug.DrawRay(hit.point, hit.normal, Color.green);

            return true;
        }

        return false;
    }
